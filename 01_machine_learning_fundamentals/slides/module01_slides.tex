      \end{itemize}
    \end{column}
  \end{columns}

  \nvbox{Intinya}{Machine learning adalah program yang aturannya dipelajari dari contoh,

% ----------------------------------------------------------
% FRAME 15 — Loss regresi
% ----------------------------------------------------------
\begin{frame}{Loss: satu angka yang diminimalkan}
  \vspace{0.8em}
  \begin{center}
    \nvbox{Mean Squared Error}{
      \centerline{$\displaystyle \text{MSE} \;=\; \frac{1}{n}\sum_{i=1}^{n}\bigl(y_i - \hat{y}_i\bigr)^2$}
    }
  \end{center}
  \vspace{0.8em}
  \begin{itemize}\setlength\itemsep{0.45em}
    \item $y_i$ nilai sebenarnya, $\hat{y}_i$ prediksi model, $n$ jumlah baris.
    \item Selisihnya dikuadratkan, jadi error besar menyumbang jauh lebih banyak daripada
          beberapa error kecil.
    \item Pelatihan mencari parameter dengan MSE terkecil di data train. Angka yang
          dilaporkan tetap diambil dari data yang tidak dipakai melatih.
  \end{itemize}
\end{frame}

% FRAME 26 — Mekanisme linear regression
% ----------------------------------------------------------
\begin{frame}{Linear regression: satu garis, dua parameter}
  \vspace{0.5em}
  \begin{center}
    \nvbox{Bentuk modelnya}{
      \centerline{$\displaystyle \hat{y} \;=\; w\,x + b$}
    }
  \end{center}
  \vspace{0.7em}
  \begin{itemize}\setlength\itemsep{0.45em}
    \item $w$ adalah kemiringan (slope): berapa banyak prediksi berubah tiap satu satuan
          kenaikan $x$.
    \item $b$ adalah intercept: nilai prediksi saat $x = 0$.
    \item Dengan banyak feature, tiap feature punya satu $w$ sendiri:
          $\hat{y} = w_1 x_1 + w_2 x_2 + \dots + w_k x_k + b$.
    \item Pelatihan memilih $w$ dan $b$ yang membuat MSE di data train terkecil.
  \end{itemize}
\end{frame}

% FRAME 27 — Kapan pakai & jebakan linear regression
% ----------------------------------------------------------
\begin{frame}{Linear regression: kapan pakai dan jebakannya}
  \vspace{0.4em}
  \kapan{
    \item hubungan feature dan target mendekati lurus
    \item kamu perlu model yang koefisiennya bisa dibaca dan dijelaskan
    \item jumlah feature masih sedikit
  }{
    \item polanya jelas melengkung atau berjenjang
    \item ada outlier (pencilan) yang menarik garisnya
    \item pengaruh satu feature bergantung pada feature lain
  }
  \vspace{1.0em}
  \jebakan{koefisien besar tidak berarti feature itu penyebabnya. Selain itu $x = 0$ sering
  berada di luar rentang data, jadi intercept belum tentu punya arti nyata.}
\end{frame}

\begin{frame}{Regularisasi: membatasi ukuran koefisien}
  \vspace{0.4em}
  {\small Regularisasi (regularization) membatasi kompleksitas model agar tidak terlalu
  menyesuaikan diri dengan data train.}
  \vspace{0.5em}
  \begin{center}
    \nvbox{Yang diminimalkan}{
      \centerline{loss di data train \;$+$\; penalti atas ukuran koefisien}
    }
  \end{center}
  \vspace{0.5em}
  \begin{itemize}\setlength\itemsep{0.4em}
    \item \textbf{\color{nvgreen}Ridge} mengecilkan semua koefisien, jarang sampai nol.
    \item \textbf{\color{nvlightgreen}Lasso} bisa menolkan koefisien yang tidak berguna,

    \item Kekuatan penaltinya dipilih dengan cross-validation. Penalti yang terlalu kuat
          membuat model melewatkan pola penting.
  \end{itemize}
\end{frame}

% FRAME 37 — Mekanisme logistic regression
% ----------------------------------------------------------
\begin{frame}{Logistic regression: threshold dan data timpang}
  \vspace{0.5em}
  \begin{itemize}\setlength\itemsep{0.45em}
    \item Keluaran modelnya probabilitas. Label kelas muncul setelah probabilitas itu
          dibandingkan dengan sebuah threshold; \texttt{predict()} memakai 0,5.
    \item Menurunkan threshold menaikkan recall dan biasanya menurunkan precision.
          Menaikkannya memberi efek sebaliknya. Jadi threshold adalah pilihan, bukan angka baku.
    \item Pilih thresholdnya dari data validation, mengikuti biaya kesalahan yang sudah
          disepakati.
    \item Kalau satu kelas jauh lebih banyak, \texttt{class\_weight='balanced'} memperbesar
          bobot kelas minoritas saat melatih model.
  \end{itemize}
\end{frame}

% ----------------------------------------------------------
% FRAME 38 — Kapan pakai & jebakan logistic
% ----------------------------------------------------------
\begin{frame}{Logistic regression: kapan pakai dan jebakannya}
  \vspace{0.4em}
  \kapan{
    \item kamu butuh probabilitas, bukan hanya label
    \item pengaruh tiap feature perlu bisa dijelaskan
    \item batas antar kelas mendekati garis lurus
  }{
    \item batas antar kelas jelas melengkung
    \item banyak interaksi antar feature yang tidak kamu buat sendiri
    \item skala feature dibiarkan berbeda-beda tanpa scaling
  }
  \vspace{1.0em}
  \jebakan{akurasi 89\% pada data yang 89\% berlabel ``tidak'' sama saja dengan baseline
  kelas mayoritas. Lihat precision, recall, dan ROC-AUC sebelum menyimpulkan.}
\end{frame}

% FRAME 40 — Mekanisme decision tree
% ----------------------------------------------------------
\begin{frame}{Decision tree: bagaimana pertanyaannya dipilih}
  \vspace{0.5em}
  \begin{itemize}\setlength\itemsep{0.45em}
    \item Di tiap simpul, model mencoba banyak pasangan feature dan ambang, lalu memilih
          pasangan yang paling memisahkan kelas.
    \item Ukuran pemisahannya bisa dipilih; ukuran default di scikit-learn adalah
          \textbf{gini}: makin kecil, makin murni isi tiap cabang.
    \item Proses itu diulang pada tiap cabang sampai simpulnya murni atau batas yang kamu
          pasang tercapai.
    \item Batasnya yang penting: \texttt{max\_depth} (kedalaman) dan
          \texttt{min\_samples\_leaf} (isi minimum tiap ujung).
  \end{itemize}
  \vspace{0.6em}
  {\footnotesize\color{nvgray}Karena tiap pertanyaan hanya membandingkan satu feature dengan satu
  angka, batas keputusannya selalu sejajar sumbu feature.}
\end{frame}

% ----------------------------------------------------------
% FRAME 41 — Kapan pakai & jebakan decision tree
% ----------------------------------------------------------
\begin{frame}{Decision tree: kapan pakai dan jebakannya}
  \vspace{0.4em}
  \kapan{
    \item aturan hasilnya perlu dibaca orang lain
    \item feature bercampur angka dan kategori
    \item kamu ingin melewati scaling; pohon tidak membutuhkannya
  }{
    \item batas antar kelas miring atau melengkung halus
    \item datanya sedikit dan pohonnya dibiarkan tumbuh bebas
    \item kamu butuh prediksi yang stabil terhadap perubahan kecil data
  }
  \vspace{1.0em}
  \jebakan{tanpa batas kedalaman, pohon dapat menghafal data train sampai errornya nol dan
  hasilnya jatuh di data baru. Pilih kedalamannya dengan cross-validation.}
\end{frame}

% FRAME 43 — Mekanisme KNN
% ----------------------------------------------------------
\begin{frame}{KNN: jarak, K, dan suara mayoritas}
  \vspace{0.1em}
  \begin{center}
    \nvbox{Jarak Euclidean antara dua baris data}{
      \centerline{$\displaystyle d(a,b) \;=\; \sqrt{\sum_{j=1}^{m}\bigl(a_j - b_j\bigr)^2}$}
    }
  \end{center}
  \vspace{0.1em}
  \begin{itemize}\setlength\itemsep{0.1em}
    \item $a$ dan $b$ dua baris data, $m$ jumlah feature, dan $K$ jumlah tetangga yang dilihat.
    \item Hitung jarak data baru ke data train, ambil K yang terdekat, lalu pakai kelas
          terbanyak di antara mereka.
    \item K kecil membuat prediksi mengikuti tiap variasi acak di sekitarnya. K besar membuat
          batasnya terlalu rata dan pola lokal hilang.
    \item Tidak ada parameter yang dilatih. Hampir semua pekerjaan terjadi saat prediksi.
  \end{itemize}
\end{frame}

% ----------------------------------------------------------
% FRAME 44 — Kapan pakai & jebakan KNN
% ----------------------------------------------------------
\begin{frame}{KNN: kapan layak dicoba}
  \vspace{0.4em}
  \begin{itemize}\setlength\itemsep{0.6em}
    \item Mulai dari data kecil dengan sedikit feature yang relevan.
    \item Prediksi dapat melambat saat data train membesar, karena pencarian tetangga
          dilakukan setiap kali memprediksi.
    \item Jika skala feature berbeda, lakukan scaling agar perhitungan jarak tidak didominasi
          satu feature.
  \end{itemize}
  \vspace{0.7em}
  \jebakan{gaji ditulis dalam juta rupiah dan umur dalam tahun. Tanpa scaling, selisih gaji
  yang rentang nilainya jauh lebih besar hampir menentukan seluruh jaraknya.}
\end{frame}

% FRAME 46 — Mekanisme SVM
% ----------------------------------------------------------
\begin{frame}{SVM: margin, C, dan kernel}
  \vspace{0.5em}
  \begin{itemize}\setlength\itemsep{0.45em}
    \item \textbf{Margin} adalah lebar jalan antara batas keputusan dan support vector
          terdekat. Pelatihan memaksimalkan lebar itu.
    \item \textbf{C} mengatur seberapa banyak titik boleh masuk ke dalam margin. C kecil
          memberi margin lebar dengan beberapa pelanggaran; C besar memaksa margin sempit yang
          rapat mengikuti data train.
    \item \textbf{Kernel RBF} mengangkat data ke ruang berdimensi lebih tinggi, tempat kedua
          kelas dapat dipisahkan bidang datar. Kembali ke dua feature semula, batasnya terlihat
          melengkung.
    \item \texttt{gamma} pada RBF mengatur seberapa lokal pengaruh satu titik.
  \end{itemize}
\end{frame}

    \item hasilnya harus dijelaskan sebagai aturan per feature

  \vspace{1.0em}
  \jebakan{kernel RBF melambat cepat saat data membesar, dan SVM mengukur jarak. Jika skala
  feature berbeda, lakukan scaling lebih dulu.}
\end{frame}

\begin{frame}{Dua cara menggabungkan, dua masalah yang berbeda}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\color{nvgreen}Bagging: menurunkan variance}
      \begin{itemize}\setlength\itemsep{0.35em}
        \item Banyak model dilatih secara terpisah pada data yang sedikit berbeda.
        \item Masing-masing tidak stabil, tetapi rata-ratanya jauh lebih tenang.

      \begin{itemize}\setlength\itemsep{0.35em}

        \item Pola yang tadinya terlewat mulai tertangkap.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{1.0em}
  \nvbox{Batasnya}{Bagging tidak banyak menolong kalau modelnya memang terlalu sederhana untuk
  datanya. Boosting yang dibiarkan menambah model terus akhirnya mengikuti noise di data train,
  jadi jumlah modelnya perlu dibatasi dari hasil di data validation.}

\begin{frame}{Hard voting dan soft voting}
  \vspace{0.3em}
  \begin{itemize}\setlength\itemsep{0.4em}
    \item \textbf{Hard voting}: setiap model memberi satu suara berupa kelas, lalu kelas dengan
          suara terbanyak dipakai.
    \item \textbf{Soft voting}: probabilitas tiap kelas dari semua model dirata-ratakan, lalu
          kelas dengan rata-rata tertinggi dipakai. Ini hanya bisa kalau setiap model memang
          mengeluarkan probabilitas.
    \item Soft voting sering sedikit lebih baik karena model yang ragu-ragu tidak berbobot sama
          dengan model yang yakin. Selisihnya tetap perlu dicek di data validation.
  \end{itemize}
  \vspace{1.0em}
  \jebakan{tiga model yang mirip menghasilkan kesalahan di baris yang sama, jadi voting-nya
  tidak menambah apa-apa. Yang menolong adalah model yang cara kerjanya berbeda.}
\end{frame}

% FRAME 68 — Mekanisme k-means
% ----------------------------------------------------------
\begin{frame}{K-Means: empat langkah yang diulang}
  \vspace{0.4em}
  \begin{enumerate}\setlength\itemsep{0.4em}
    \item Tentukan K, lalu tempatkan K pusat awal.
    \item Kelompokkan setiap baris ke pusat yang paling dekat.
    \item Hitung ulang setiap pusat sebagai rata-rata anggota kelompoknya.
    \item Ulangi langkah 2 dan 3 sampai keanggotaannya tidak berubah lagi.
  \end{enumerate}
  \vspace{0.9em}
  \nvbox{Pusat awal ikut menentukan hasil}{Dari pusat awal yang berbeda, prosesnya bisa berhenti
  di kelompok yang berbeda. \texttt{n\_init} di scikit-learn menjalankan seluruh proses beberapa
  kali dengan pusat awal berbeda, lalu mengambil hasil dengan inertia terkecil.}
\end{frame}

% FRAME 69 — Kapan pakai & jebakan k-means
% ----------------------------------------------------------
\begin{frame}{K-Means: kapan pakai dan jebakannya}
  \vspace{0.4em}
  \kapan{
    \item ukuran kelompoknya mirip dan bentuknya membulat
    \item semua feature berupa angka dengan skala setara
    \item K bisa ditentukan dari kebutuhan, atau dicari
  }{
    \item bentuk kelompoknya memanjang atau melengkung
    \item sebagian feature berupa kategori
    \item ada outlier (pencilan) yang menarik pusat menjauh
  }
  \vspace{1.0em}
  \jebakan{k-means bekerja dengan jarak, jadi kelompok yang keluar cenderung bulat dan
  seukuran. Kolom kategori tidak punya arti jarak, dan kolom belanja yang sebarannya miring
  perlu diubah dengan \texttt{log1p} lebih dulu.}

% ----------------------------------------------------------
% FRAME 78 — Apa yang cocok di GPU
% ----------------------------------------------------------
\begin{frame}{Pekerjaan seperti apa yang cocok di GPU}
  \vspace{0.4em}
  \kapan{
    \item operasinya sama untuk semua baris, misalnya perkalian matriks atau perhitungan jarak
    \item barisnya banyak, sehingga semua core punya bagian
    \item datanya sudah berada di memori GPU
  }{
    \item datanya kecil, hanya beberapa ribu baris
    \item alurnya bercabang berbeda-beda per baris
    \item data harus bolak-balik dipindah antara CPU dan GPU
  }
  \vspace{0.9em}
  \nvbox{Yang dijalankan di GPU pada nb10}{Random Forest dan KNN dari cuML, K-Means, serta
  XGBoost dengan \texttt{device="cuda"}.}
\end{frame}

% ----------------------------------------------------------
% FRAME 79 — cuML
% ----------------------------------------------------------
\begin{frame}[fragile]{cuML: yang berubah hanya baris import-nya}
  \vspace{0.3em}
\begin{lstlisting}[language=Python, numbers=none, basicstyle=\ttfamily\small\color{nvwhite}]
# CPU -- scikit-learn
from sklearn.ensemble import RandomForestClassifier

# GPU -- cuML (bagian dari RAPIDS)
from cuml.ensemble import RandomForestClassifier

model = RandomForestClassifier(n_estimators=200)
model.fit(X_train, y_train)
pred = model.predict(X_test)
\end{lstlisting}
  \vspace{0.3em}
  \begin{itemize}\setlength\itemsep{0.3em}
    \item Nama kelas dan urutan pemanggilannya sama, jadi kode setelah import tidak perlu diubah.
    \item Tidak semua argumen scikit-learn tersedia di cuML, dan hasilnya bisa berbeda sedikit
          karena urutan perhitungannya berbeda.
  \end{itemize}
\end{frame}

% ----------------------------------------------------------
% FRAME 80 — XGBoost di GPU
% ----------------------------------------------------------
\begin{frame}[fragile]{XGBoost di GPU: satu argumen}
  \vspace{0.5em}
\begin{lstlisting}[language=Python, numbers=none, basicstyle=\ttfamily\normalsize\color{nvwhite}]
model_cpu = XGBClassifier(n_estimators=300, device="cpu")

model_gpu = XGBClassifier(n_estimators=300, device="cuda")
\end{lstlisting}
  \vspace{0.7em}
  \begin{itemize}\setlength\itemsep{0.4em}
    \item Sisa kodenya sama: \texttt{fit()}, \texttt{predict()}, dan metriknya tidak berubah.
    \item Kalau GPU tidak tersedia, \texttt{device="cuda"} akan gagal atau jatuh kembali ke CPU,
          jadi ketersediaan GPU perlu dicek lebih dulu.
    \item Hasil prediksinya bisa berbeda sedikit dari versi CPU karena urutan penjumlahan
          bilangan pecahan berbeda.
  \end{itemize}
\end{frame}

% ----------------------------------------------------------
% FRAME 81 — Kapan GPU tidak menang
% ----------------------------------------------------------
\begin{frame}{Kapan GPU tidak menang}
  \vspace{0.5em}
  \begin{itemize}\setlength\itemsep{0.5em}
    \item Data harus dipindah ke memori GPU lebih dulu. Kalau pekerjaannya singkat, waktu
          pindah itu lebih besar daripada waktu yang dihemat.
    \item Panggilan pertama memuat kernel dan mengalokasikan memori, jadi selalu terasa lambat.
          Itu warm-up, dan bukan bagian yang diukur.
    \item Pada data kecil, satu core CPU yang kuat sering lebih cepat daripada ribuan core yang
          menunggu data.
    \item Pekerjaan yang alurnya bercabang berbeda per baris tidak bisa dibagi rata ke banyak
          core.
  \end{itemize}
  \vspace{0.7em}
  {\footnotesize\color{nvgray}Urutan yang benar saat mengukur: jalankan sekali untuk warm-up,
  baru catat waktunya.}
\end{frame}
