% FRAME 8 — Loss
% ----------------------------------------------------------
\begin{frame}{Loss: satu angka yang mengukur seberapa salah}
  \vspace{0.5em}
  \begin{center}
    \color{nvlightgreen}$\displaystyle \text{loss satu contoh} = -\log\left(p_{\text{kelas benar}}\right)$
  \end{center}
  \vspace{0.5em}
  \begin{itemize}\setlength\itemsep{0.42em}
    \item Model mengeluarkan satu probabilitas untuk tiap kelas. Loss hanya melihat
          probabilitas yang diberikan ke kelas yang benar.
    \item Probabilitas 0,85 untuk kelas yang benar memberi loss kecil; 0,05 memberi loss besar.
    \item Loss seluruh batch adalah rata-rata loss tiap contohnya. Angka inilah yang
          diperkecil saat training.
  \end{itemize}
  \vspace{0.5em}
  {\footnotesize\color{nvgray}Namanya di Keras: \texttt{sparse\_categorical\_crossentropy}.
  Accuracy dilaporkan untuk kita baca; loss yang dipakai model untuk memperbaiki diri.}
\end{frame}

\begin{frame}{Learning rate: ukuran langkah dan gejalanya}
  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\color{nvlightgreen}Terlalu kecil}
      \vspace{0.35em}
      \begin{itemize}\setlength\itemsep{0.38em}
        \item Loss turun, tetapi sangat lambat
        \item Panel kiri: 12 langkah belum sampai ke dasar
        \item Epoch habis sebelum modelnya selesai belajar
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\color{nvred}Terlalu besar}
      \vspace{0.35em}
      \begin{itemize}\setlength\itemsep{0.38em}
        \item Langkahnya melewati dasar, lalu menjauh
        \item Panel kanan: nilai $w$ makin jauh tiap langkah
        \item Loss naik-turun tanpa arah, atau menjadi NaN
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.9em}
  \jebakan{Adam memakai learning rate 0,001 sebagai nilai default. Itu titik awal yang
  wajar, bukan angka yang cocok untuk semua data. Kalau loss diam saja atau meledak sejak
  epoch pertama, ubah dulu learning ratenya sebelum menambah layer.}
\end{frame}

\begin{frame}{ReLU: turunannya tetap 1 di sisi positif}
  \vspace{0.2em}
  \begin{columns}[T]
    \begin{column}{0.34\textwidth}
      \centering
      \textbf{\color{nvgreen}ReLU}\\[0.3em]
      \begin{tikzpicture}[scale=0.78, font=\tiny]
        \draw[nvgray, ->] (-1.7,0) -- (1.9,0) node[below left, text=nvgray] {$x$};
        \draw[nvgray, ->] (0,-0.7) -- (0,1.9) node[left, text=nvgray] {$f(x)$};
        \draw[nvgreen, line width=1.3pt] (-1.6,0) -- (0,0) -- (1.7,1.7);
      \end{tikzpicture}\\[0.3em]
      {\scriptsize $f(x)=\max(0,\,x)$}\\[0.5em]
      \textbf{\color{nvlightgreen}Leaky ReLU}\\[0.3em]
      \begin{tikzpicture}[scale=0.78, font=\tiny]
        \draw[nvgray, ->] (-1.7,0) -- (1.9,0) node[below left, text=nvgray] {$x$};
        \draw[nvgray, ->] (0,-0.7) -- (0,1.9) node[left, text=nvgray] {$f(x)$};
        \draw[nvlightgreen, line width=1.3pt] (-1.6,-0.16) -- (0,0) -- (1.7,1.7);
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.62\textwidth}
      \vspace{0.2em}
      \begin{itemize}\setlength\itemsep{0.35em}
        \item ReLU meneruskan nilai positif apa adanya dan menolkan yang negatif.
              Turunannya 1 di sisi positif, jadi kesalahan tidak menyusut saat dibagi
              ke belakang.
        \item Perhitungannya juga murah: satu perbandingan dengan nol per neuron.
        \item Sisi negatifnya nol. Neuron yang masukannya negatif terus berhenti
              diperbarui, dan keadaan itu disebut dead ReLU.
        \item Leaky ReLU memberi kemiringan kecil di sisi negatif, 0,1 pada gambar
              sebelumnya, sehingga masih ada gradient yang mengalir.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ----------------------------------------------------------
\begin{frame}{Kapan mengganti aktivasi atau optimizer default}
  \vspace{0.3em}
  \kapan{
    \item banyak neuron ReLU berhenti aktif, keluarannya nol terus
    \item loss diam sejak epoch awal padahal datanya sudah dinormalisasi
    \item jaringannya dalam dan layer awalnya hampir tidak berubah
  }{
    \item lossnya belum turun karena learning ratenya belum disesuaikan
    \item perbandingannya belum diuji di data validation
    \item modelnya masih underfit dan layernya memang terlalu sedikit
  }
  \vspace{0.7em}
  \jebakan{Dead ReLU biasanya berawal dari learning rate yang terlalu besar, bukan dari
  ReLU itu sendiri. Kalau lossnya tidak turun, ganti learning ratenya dulu, sebelum
  arsitekturnya.}
\end{frame}

% FRAME 38 — Ringkasan Act 3
% ----------------------------------------------------------
\begin{frame}{Ringkasan Act 3}
  \vspace{0.5em}
  \begin{itemize}\setlength\itemsep{0.46em}
    \item Gambar masuk ke model sebagai tabel angka. Konvolusi memanfaatkan kedekatan antar
          pixel yang diabaikan Dense layer.
    \item Satu filter kecil digeser ke seluruh gambar dan menghasilkan satu feature map.
          Bobot yang dibagi bersama membuat parameternya jauh lebih sedikit.
    \item Pooling menyusutkan peta dengan menyimpan nilai terkuat, tanpa menambah parameter.
    \item Blok Conv $\rightarrow$ Pool ditumpuk sampai petanya kecil, lalu Flatten dan Dense
          mengambil keputusan kelas.
    \item Augmentasi sebagai layer hanya aktif saat training. Transfer learning meminjam
          backbone terlatih dan melatih kepalanya, dan keunggulannya bergantung pada
          resolusi serta kedekatan domain.
  \end{itemize}
\end{frame}

% FRAME 53 — Kapan GPU tidak untung
% ----------------------------------------------------------
\begin{frame}{Kapan GPU tidak memberi keuntungan}
  \vspace{0.2em}
  \begin{itemize}\setlength\itemsep{0.4em}
    \item \textbf{\color{nvlightgreen}Modelnya kecil.} Beberapa layer Dense di atas data
          tabular selesai sebelum GPU sempat sibuk.
    \item \textbf{\color{nvlightgreen}Batchnya kecil.} Batch 8 memberi sedikit pekerjaan per
          langkah, jadi banyak unit hitung menganggur.
    \item \textbf{\color{nvlightgreen}Datanya bolak-balik.} Setiap batch harus dipindahkan ke
          memori GPU dulu; kalau hitungannya ringan, pemindahan itu yang mendominasi.
    \item \textbf{\color{nvlightgreen}Pemanggilan pertama mahal.} Kernel dan graph disiapkan
          saat dipakai pertama kali, jadi panggilan pertama jauh lebih lambat daripada
          panggilan berikutnya.
  \end{itemize}
  \vspace{0.4em}
  \jebakan{Mengukur tanpa warm-up membuat GPU tampak lambat. Di nb05, \texttt{fit} dan
  \texttt{predict} dipanggil sekali di kedua sisi sebelum stopwatch dinyalakan, supaya yang
  dibandingkan hitungannya.}
\end{frame}

% ----------------------------------------------------------
% FRAME 54 — Mixed precision
% ----------------------------------------------------------
\begin{frame}[fragile]{Mixed precision: hitung di float16, keluaran di float32}
  \vspace{0.2em}
\begin{lstlisting}
from tensorflow.keras import mixed_precision
mixed_precision.set_global_policy('mixed_float16')
Dense(10, activation='softmax', dtype='float32')  # keluaran
\end{lstlisting}
  \vspace{0.4em}
  \begin{itemize}\setlength\itemsep{0.32em}
    \item float16 memakai 16 bit per angka, float32 memakai 32 bit. GPU NVIDIA modern punya
          Tensor Core yang mengerjakan operasi float16.
    \item Bagian yang rawan dibiarkan di float32: keluaran softmax dan penjumlahan gradient.
          Karena itu output layernya diberi \texttt{dtype='float32'}.
    \item Bukan tombol ajaib: keuntungannya terasa pada model dan batch yang cukup besar, dan
          besarnya diukur di nb05, bukan diperkirakan.
  \end{itemize}
  \vspace{0.4em}
  {\footnotesize\color{nvgray}Di nb05 policynya dikembalikan ke \texttt{float32} setelah
  percobaan, supaya sel-sel berikutnya tidak ikut terpengaruh.}
\end{frame}

\begin{frame}{Digit yang lahir dari noise, epoch demi epoch}
  \vspace{0.1em}
  \begin{center}
    \includegraphics[width=0.72\textwidth, height=2.7cm, keepaspectratio]{figures/gan_grid.pdf}
  \end{center}
  \vspace{0.1em}
  \begin{itemize}\setlength\itemsep{0.22em}
    \item DCGAN pada MNIST, 15 epoch, seed 42. Tiap baris memakai noise masukan yang sama,
          jadi yang berubah hanya generatornya: epoch 1 masih berbintik, epoch 15 sebagian
          sudah berbentuk digit yang bisa dikenali dan tidak satu pun ada di dataset.

  \jebakan{Mode collapse: generator bisa menemukan satu atau dua bentuk yang lolos terus, lalu
  berhenti belajar variasi lain. Gejalanya keluaran yang mirip satu sama lain, jadi yang
  diperiksa keragamannya.}
\end{frame}

% FRAME 67 — Kapan pakai & jebakan generatif
% ----------------------------------------------------------
\begin{frame}{Kapan model generatif cocok, dan apa jebakannya}
  \vspace{0.3em}
  \kapan{
    \item butuh contoh baru: menambah variasi data, mengisi bagian yang hilang
    \item butuh ringkasan padat: kompresi, denoising, deteksi anomali dari galat rekonstruksi
    \item hasilnya dinilai orang, bukan dicocokkan dengan satu jawaban
  }{
    \item pertanyaannya klasifikasi atau prediksi satu angka
    \item jawabannya harus bisa diperiksa benar atau salah
    \item data trainnya sedikit dan sempit
  }
  \vspace{0.5em}
  \jebakan{Model generatif menghasilkan yang wajar menurut data trainnya, belum tentu yang
  benar. Akurasi tidak bisa dipakai menilainya, dan di Modul 5 kita lihat satu cara mengikat
  keluaran ke sumber yang bisa ditunjuk.}
\end{frame}

% FRAME 68 — Ringkasan Act 6
% ----------------------------------------------------------
\begin{frame}{Ringkasan Act 6}
  \vspace{0.5em}
  \begin{itemize}\setlength\itemsep{0.46em}
    \item Model diskriminatif memetakan data ke label; model generatif belajar seperti apa data
          yang wajar, lalu menghasilkan contoh baru.
    \item Autoencoder memampatkan gambar menjadi beberapa angka dan membangunnya ulang.
          Decodernya bisa dipakai sendiri sebagai generator.
    \item Latent space menyimpan struktur, terlihat dari interpolasi yang mulus. Titik acak di
          daerah kosong masih memberi hasil kabur.
    \item GAN melatih generator dan discriminator sebagai lawan tanding. Hasilnya bisa bagus,
          tetapi rewel dan rentan mode collapse.
    \item Diffusion membalik penambahan noise selangkah demi selangkah: mahal saat training,
          bertahap saat menghasilkan gambar.
  \end{itemize}
\end{frame}
